## 🚀 Quick Start

```typescript
import { createAIEstimator } from '@orpaynter/toolkit-ai-estimator';
import { createDamageDetector } from '@orpaynter/toolkit-damage-detector';
import { createWeatherIntel } from '@orpaynter/toolkit-weather-intel';
import { createVisualizer3D } from '@orpaynter/toolkit-3d-visualizer';
import { createContractorCopilot } from '@orpaynter/toolkit-contractor-copilot';

// Initialize all toolkits
const estimator = createAIEstimator();
const detector = createDamageDetector({ enableAR: true });
const weather = createWeatherIntel();
const visualizer = createVisualizer3D({ renderQuality: 'ultra' });
const copilot = createContractorCopilot({ voiceEnabled: true });

// Example workflow
const measurements = await estimator.analyzeRoofImage(photo);
const damageReport = await detector.analyzeImage(photo);
const workWindows = await weather.findWorkWindows(7);
const render3D = await visualizer.applyMaterial(model, 'charcoal');
const schedule = await copilot.optimizeSchedule(projects);
```
